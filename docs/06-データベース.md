## 1. 目的と完成の定義

**目的**: データを永続化する。SQLの基礎、`database/sql`、マイグレーション、環境変数による設定を学ぶ。インターフェースのおかげでhandlerを変えずに差し替えられることを体験する。

**完成の定義**:
- [ ] サーバーを再起動してもデータが残る（F-08）
- [ ] `store` のSQLite実装があり、メモリ実装と**同じテスト**が通る
- [ ] マイグレーションSQLが `backend/migrations/` にある
- [ ] DBのパスは環境変数 `DB_PATH`。`.env.example` がある
- [ ] `.env` と `*.db` がコミットされていない

## 2. 用語解説

| 用語 | 意味 |
|------|------|
| SQL | データベースを操作する言語 (SELECT / INSERT / UPDATE / DELETE) |
| マイグレーション | テーブル構造の変更履歴をファイルで管理すること |
| プレースホルダ | `?` で値を渡す書き方。**SQLインジェクション対策** |
| `database/sql` | Go標準のDB操作API。ドライバーを別途読み込む |
| トランザクション | 複数の操作を「全部成功か全部失敗」にする仕組み |
| `.env.example` | 値を空/ダミーにした設定の見本（コミットする） |

- SQL基礎を軽く学ぶ（SELECT / WHERE / ORDER BY / INSERT / UPDATE / DELETE）。AIに「SQLiteの基礎を3分で」と頼んでもよい

## 6. つまずきやすい点
- `database is locked` → `rows.Close()` の閉じ忘れ。`defer rows.Close()`
- 日時の保存形式がバラバラ → UTCのISO 8601で統一
- `SELECT *` を避け、カラムを明示する
- `done` のbool変換（0/1 ⇔ true/false）を忘れる
- 相対パスの `DB_PATH` は**実行ディレクトリ**基準

ドライバー追加:


`backend/migrations/migrations.go`（SQLファイルをバイナリに埋め込む）
```go
package migrations

import "embed"

//go:embed *.sql
var FS embed.FS
```
※ `embed` は自分より**下**のファイルしか埋め込めないため、SQLと同じフォルダにGoファイルを置く。

`backend/internal/store/sqlite.go`（接続部分）
```go
package store

import (
	"context"
	"database/sql"
	"sort"

	_ "modernc.org/sqlite" // ドライバー登録のためだけにimport

	"todoapp/backend/migrations"
)

type SQLite struct {
	db *sql.DB
}

func NewSQLite(path string) (*SQLite, error) {
	db, err := sql.Open("sqlite", path)
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(1) // SQLiteは書き込みが1つずつ。"database is locked" を避ける
	if err := db.Ping(); err != nil {
		db.Close()
		return nil, err
	}
	return &SQLite{db: db}, nil
}

func (s *SQLite) Close() error { return s.db.Close() }

// Migrate は migrations/*.sql をファイル名順に実行する。
func (s *SQLite) Migrate(ctx context.Context) error {
	entries, err := migrations.FS.ReadDir(".")
	if err != nil {
		return err
	}
	names := make([]string, 0, len(entries))
	for _, e := range entries {
		names = append(names, e.Name())
	}
	sort.Strings(names)

	for _, name := range names {
		b, err := migrations.FS.ReadFile(name)
		if err != nil {
			return err
		}
		if _, err := s.db.ExecContext(ctx, string(b)); err != nil {
			return err
		}
	}
	return nil
}
```

<details>
<summary>Step 4 の答え: SQLite実装（sqlite.go の続き）</summary>

importに `"strings"` `"time"` `"errors"` `"todoapp/backend/internal/model"` を追加。

```go
const columns = "id, title, done, created_at, updated_at"

// *sql.Row と *sql.Rows の両方で使えるようにする
type scanner interface {
	Scan(dest ...any) error
}

func scanTodo(sc scanner) (model.Todo, error) {
	var t model.Todo
	var done int
	var created, updated string
	if err := sc.Scan(&t.ID, &t.Title, &done, &created, &updated); err != nil {
		return model.Todo{}, err
	}
	t.Done = done == 1
	var err error
	if t.CreatedAt, err = time.Parse(time.RFC3339, created); err != nil {
		return model.Todo{}, err
	}
	if t.UpdatedAt, err = time.Parse(time.RFC3339, updated); err != nil {
		return model.Todo{}, err
	}
	return t, nil
}

func now() string { return time.Now().UTC().Format(time.RFC3339) }

func (s *SQLite) List(ctx context.Context, status string) ([]model.Todo, error) {
	q := "SELECT " + columns + " FROM todos" // 連結しているのは固定文字列のみ
	switch status {
	case "active":
		q += " WHERE done = 0"
	case "done":
		q += " WHERE done = 1"
	}
	q += " ORDER BY created_at DESC, id DESC"

	rows, err := s.db.QueryContext(ctx, q)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	todos := []model.Todo{} // 0件でも [] にする
	for rows.Next() {
		t, err := scanTodo(rows)
		if err != nil {
			return nil, err
		}
		todos = append(todos, t)
	}
	return todos, rows.Err()
}

func (s *SQLite) Get(ctx context.Context, id int64) (model.Todo, error) {
	row := s.db.QueryRowContext(ctx, "SELECT "+columns+" FROM todos WHERE id = ?", id)
	t, err := scanTodo(row)
	if errors.Is(err, sql.ErrNoRows) {
		return model.Todo{}, ErrNotFound
	}
	return t, err
}

func (s *SQLite) Create(ctx context.Context, title string) (model.Todo, error) {
	ts := now()
	res, err := s.db.ExecContext(ctx,
		"INSERT INTO todos (title, done, created_at, updated_at) VALUES (?, 0, ?, ?)",
		title, ts, ts)
	if err != nil {
		return model.Todo{}, err
	}
	id, err := res.LastInsertId()
	if err != nil {
		return model.Todo{}, err
	}
	return s.Get(ctx, id)
}

func (s *SQLite) Update(ctx context.Context, id int64, in UpdateInput) (model.Todo, error) {
	sets := []string{"updated_at = ?"}
	args := []any{now()}
	if in.Title != nil {
		sets = append(sets, "title = ?")
		args = append(args, *in.Title)
	}
	if in.Done != nil {
		done := 0
		if *in.Done {
			done = 1
		}
		sets = append(sets, "done = ?")
		args = append(args, done)
	}
	args = append(args, id)

	res, err := s.db.ExecContext(ctx,
		"UPDATE todos SET "+strings.Join(sets, ", ")+" WHERE id = ?", args...)
	if err != nil {
		return model.Todo{}, err
	}
	if n, _ := res.RowsAffected(); n == 0 {
		return model.Todo{}, ErrNotFound
	}
	return s.Get(ctx, id)
}

func (s *SQLite) Delete(ctx context.Context, id int64) error {
	res, err := s.db.ExecContext(ctx, "DELETE FROM todos WHERE id = ?", id)
	if err != nil {
		return err
	}
	if n, _ := res.RowsAffected(); n == 0 {
		return ErrNotFound
	}
	return nil
}
```
- 値はすべて `?` で渡している（SQLインジェクション対策）。連結しているのは、プログラムが決めた固定文字列だけ
- `sql.ErrNoRows` を `ErrNotFound` に変換することで、handlerはDBの種類を知らずに済む

<summary>Step 5 の答え: メモリ版とSQLite版で同じテストを走らせる</summary>

`store_test.go`
```go
package store

import (
	"context"
	"errors"
	"path/filepath"
	"testing"
)

func newMemory(t *testing.T) TodoStore { return NewMemory() }

func newSQLite(t *testing.T) TodoStore {
	t.Helper()
	s, err := NewSQLite(filepath.Join(t.TempDir(), "test.db"))
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { s.Close() })
	if err := s.Migrate(context.Background()); err != nil {
		t.Fatal(err)
	}
	return s
}

func TestStores(t *testing.T) {
	impls := map[string]func(*testing.T) TodoStore{
		"memory": newMemory,
		"sqlite": newSQLite,
	}
	for name, newStore := range impls {
		t.Run(name, func(t *testing.T) {
			testCRUD(t, newStore(t))
		})
	}
}

func testCRUD(t *testing.T, s TodoStore) {
	ctx := context.Background()

	created, err := s.Create(ctx, "牛乳")
	if err != nil {
		t.Fatal(err)
	}
	got, err := s.Get(ctx, created.ID)
	if err != nil || got.Title != "牛乳" || got.Done {
		t.Fatalf("Get: %+v, %v", got, err)
	}

	done := true
	updated, err := s.Update(ctx, created.ID, UpdateInput{Done: &done})
	if err != nil || !updated.Done || updated.Title != "牛乳" {
		t.Fatalf("Update: %+v, %v", updated, err)
	}

	list, err := s.List(ctx, "done")
	if err != nil || len(list) != 1 {
		t.Fatalf("List done: %+v, %v", list, err)
	}
	list, _ = s.List(ctx, "active")
	if len(list) != 0 {
		t.Fatalf("List active: %+v", list)
	}

	if err := s.Delete(ctx, created.ID); err != nil {
		t.Fatal(err)
	}
	if _, err := s.Get(ctx, created.ID); !errors.Is(err, ErrNotFound) {
		t.Fatalf("want ErrNotFound, got %v", err)
	}
}
```
前の `memory_test.go` と関数名が重ならないように注意する。

```go
func main() {
	port := os.Getenv("PORT")
	if port == "" {
		port = "8080"
	}
	dbPath := os.Getenv("DB_PATH")
	if dbPath == "" {
		dbPath = "./todo.db"
	}

	st, err := store.NewSQLite(dbPath)
	if err != nil {
		log.Fatalf("open db: %v", err)
	}
	defer st.Close()
	if err := st.Migrate(context.Background()); err != nil {
		log.Fatalf("migrate: %v", err)
	}

	mux := http.NewServeMux()
	handler.NewTodoHandler(st).Register(mux) // handler は変更なし

	log.Printf("server started on :%s (db: %s)", port, dbPath)
	log.Fatal(http.ListenAndServe(":"+port, mux))
}
```
`log.Fatal` は `defer` を実行せずに終了する点に注意（学習段階では許容）。
